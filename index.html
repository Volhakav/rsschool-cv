Code Example

Solution to this Codewars kata: 


        function findOutlier(integers){
            let numOfOdd = 0;
            let numOfEven = 0;
            for(let i = 0; i < integers.length; i++){
                if(integers[i] % 2 == 0){
                    numOfEven++;
                } else {
                    numOfOdd++;
                }
            }
            if(numOfOdd < numOfEven){
                return integers.find(x => x % 2 !== 0);
            } else {
                return integers.find(x => x % 2 == 0);
            }
        }
